cmake_minimum_required(VERSION 3.20)
project(ndi_ptz_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED NDI_SDK_DIR)
 message(FATAL_ERROR "Set NDI_SDK_DIR to the NDI SDK directory")
endif()
add_executable(ndi-ptz-bridge main.cpp)
target_include_directories(ndi-ptz-bridge PRIVATE "${NDI_SDK_DIR}/Include")
if(WIN32)
 target_link_libraries(ndi-ptz-bridge PRIVATE "${NDI_SDK_DIR}/Lib/x64/Processing.NDI.Lib.x64.lib")
endif()
set_target_properties(ndi-ptz-bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
